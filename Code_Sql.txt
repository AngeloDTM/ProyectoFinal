CREATE DATABASE IF NOT EXISTS colmena CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE colmena;

CREATE TABLE IF NOT EXISTS usuarios ( id INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY, nombre VARCHAR(60) NOT NULL, usuario VARCHAR(30) NOT NULL UNIQUE, correo VARCHAR(120) NOT NULL UNIQUE, password_hash VARCHAR(255) NOT NULL, creado_en TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ) ENGINE=InnoDB;
